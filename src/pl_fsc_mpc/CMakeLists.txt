cmake_minimum_required(VERSION 3.14)
project(pl_fsc_mpc)

set(CMAKE_CXX_STANDARD 20)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(eigen3_cmake_module REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(px4_ros2_cpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)

add_subdirectory(../../third_party/fsc_ibvs fsc_ibvs)
add_executable(precision_land ../pl_rpg_mpc/PrecisionLand.cpp FscSolver.cpp)
target_include_directories(precision_land PRIVATE ../pl_rpg_mpc)
ament_target_dependencies(precision_land rclcpp Eigen3 px4_ros2_cpp geometry_msgs sensor_msgs std_srvs std_msgs)
target_link_libraries(precision_land fsc_ibvs_solver)
install(TARGETS precision_land DESTINATION lib/${PROJECT_NAME})
ament_package()
